Option Explicit

Sub Copiercollerstype(Buttonname As String)

If Buttonname = "" Then
Exit Sub
End If

Application.ScreenUpdating = False
constantes_generales
Dim col_select, row_select, row_medecin, Week_day, Week_num, col_debut As Long

col_select = ActiveCell.Column
row_select = ActiveCell.Row
Week_day = Weekday(CDate(Range("C" & row_select).Value), 2) 'Jour de la semaine
Week_num = Format(G_sh(1).Range("C" & row_select).Value, "ww", vbMonday, vbFirstFourDays) 'Numéro de semaine
row_medecin = Application.Match(Buttonname, G_sh(3).Range("C1:C30"), 0) 'Numéro ligne du médecin dans la feuille "Liste medecins"

'Vérifier si semaine paire ou impaire
If Week_num Mod 2 = 0 Then 'Si semaine paire  'colonne début dans Liste medecin = N (Soit 14)
col_debut = 14
Else 'Si semaine impaire 'Colonne début dans liste medecins = X soit 24
col_debut = 24
End If

'On applique à partir du tableau dans Liste medecins, après avoir vérifié si jour férié
If col_select > 3 And col_select < 21 Then
For i = Week_day To 5
If Not IsError(Application.Match(G_sh(1).Cells(row_select + i - Week_day, 3), Worksheets("Paramètres").Range("R2:R14"), 0)) Then
G_sh(1).Cells(row_select + i - Week_day, col_select).Value = ""
G_sh(1).Cells(row_select + i - Week_day, col_select + 1).Value = ""
Else
G_sh(1).Cells(row_select + i - Week_day, col_select).Value = G_sh(3).Cells(row_medecin, col_debut + 2 * (i - 1)).Value
G_sh(1).Cells(row_select + i - Week_day, col_select + 1).Value = G_sh(3).Cells(row_medecin, col_debut + 2 * (i - 1) + 1).Value
End If
Next i
ElseIf col_select > 47 Then
For i = Week_day To 5
G_sh(1).Cells(row_select + i - Week_day, col_select).Value = Buttonname
Next i
End If
Application.ScreenUpdating = True

End Sub

Option Explicit

Public Sub changeyear(new_year As Long)

constantes_generales
Dim addr As String

Application.ScreenUpdating = False
Stop_protect
Application.EnableEvents = False

addr = ThisWorkbook.Path & "\" & "Planning séniors " & new_year & ".xlsm"

'Vérification si le planning existe déjà dans le répertoire
If Not Dir(addr) = "" Then
MsgBox "Il existe déjà un planning pour cette année là dans le dossier. Merci de le déplacer ou le supprimer au préalable puis rappuyez."
GoTo fin
End If

'On applique la nouvelle année
G_sh(2).Range("R16").Value = new_year

constantes_generales

G_sh(1).Activate
G_sh(1).Select

G_sh(1).Range("C374").Value = ""
G_sh(1).Range("A374:CC374").Select
    Selection.Borders(xlEdgeLeft).LineStyle = xlNone
    Selection.Borders(xlEdgeTop).LineStyle = xlNone
    Selection.Borders(xlEdgeBottom).LineStyle = xlNone
    Selection.Borders(xlEdgeRight).LineStyle = xlNone
    Selection.Borders(xlInsideVertical).LineStyle = xlNone
    Selection.Borders(xlInsideHorizontal).LineStyle = xlNone
    With Selection.Interior
        .PatternColorIndex = xlAutomatic
        .ThemeColor = xlThemeColorDark1
        .TintAndShade = 0
        .PatternTintAndShade = 0
    End With
    
'Suppression des données actuelles
G_sh(1).Range("D9:CL374").Select
Selection.ClearContents

'Ajout des dates dans la colonne C
G_sh(1).Range("C9").Value = "01/01/" & new_year
For i = 1 To G_sh(2).Range("R17").Value - 1
G_sh(1).Range("C" & (9 + i)).Value = DateAdd("d", 1, G_sh(1).Range("C" & (8 + i)).Value)
Next i

'Suppression fusions cellules colonne A
For i = 9 To 374
If G_sh(1).Range("A" & i).MergeCells = True Then
G_sh(1).Range("A" & i).UnMerge
End If
Next i

'On crée les nouvelles fusions pour la colonne A
Application.DisplayAlerts = False
For i = 1 To 12
G_sh(1).Range("A" & G_firstrow(i)).Value = Worksheets("Paramètres").Range("W" & i + 1).Value
G_sh(1).Range("A" & G_firstrow(i) + 15).Value = Worksheets("Paramètres").Range("W" & i + 1).Value
            With G_sh(1).Range("A" & G_firstrow(i) & ":A" & G_lastrow(i))
                .Borders(xlEdgeTop).Weight = xlThick
                .Borders(xlEdgeBottom).Weight = xlThick
                .Borders(xlEdgeRight).Weight = xlMedium
            End With
G_sh(1).Range("A" & G_firstrow(i) & ":A" & G_firstrow(i) + 14).Merge
G_sh(1).Range("A" & G_firstrow(i) + 15 & ":A" & G_lastrow(i)).Merge
Next i
Application.DisplayAlerts = True

Calendrier_MiseEnForme

ThisWorkbook.SaveAs addr

fin:
Application.EnableEvents = True
Start_protect
Application.ScreenUpdating = True
End Sub

Public Function formatconditions(cell_cond As String, choixsheet As String, bordures_none As Integer, priority As Integer)
Dim choose_sh As Worksheet

If choixsheet = 0 Then
Set choose_sh = G_sh(3)
ElseIf choixsheet = 1 Then
Set choose_sh = G_sh(4)
ElseIf choixsheet = 2 Then
Set choose_sh = G_sh(2)
End If

With Selection.formatconditions(Selection.formatconditions.Count)

in file: xl/vbaProject.bin - OLE stream: 'VBA/Astreintes_presences'

Option Explicit

Public Sub Maj_recap_astreintes()

Application.ScreenUpdating = False
Stop_protect

Dim rng, cell, C, A, B, range_initiales As Range
Dim firstblank, col_select, row_select, Row_Previous_Saturday, Row_Next_Sunday, joursemaine As Long
Dim initiales, pref As String

col_select = ActiveCell.Column
row_select = ActiveCell.Row

constantes_generales

For u = 1 To 18
    UserFormAstreintes.Controls("commandbutton" & u).Caption = ""
    UserFormAstreintes.Controls("TextBoxast" & u).Value = ""
    UserFormAstreintes.TextBoxjour.Value = ""
    UserFormAstreintes.Controls("commandbutton" & u).BackColor = RGB(255, 255, 255)
Next u

Set rng = Range("Tableau19").ListObject.ListColumns(1).DataBodyRange
On Error Resume Next
Set cell = rng.SpecialCells(xlCellTypeBlanks)(1)
    firstblank = cell.Row
On Error GoTo 0
Set rng = Nothing: Set cell = Nothing

UserFormAstreintes.ListBox2.List = Range("Récapitulatif!B1:F" & (firstblank - 1)).Value

If Not col_select = 22 Then
GoTo fin
End If

'On récupère le jour de la semaine
joursemaine = Weekday(G_sh(1).Range("C" & row_select).Value)

If joursemaine = 1 Or joursemaine = 6 Or joursemaine = 7 Then
GoTo fin

Else
Row_Previous_Saturday = row_select - joursemaine
Row_Next_Sunday = Row_Previous_Saturday + 8

'Affichage du jour sélectionné
UserFormAstreintes.TextBoxjour.Value = Format(G_sh(1).Range("C" & row_select).Value, "dddd") & " " & G_sh(1).Range("C" & row_select).Value

'On recherche si médecin en vacances ou si médecins déjà d'astreintes (WE autour et semaine en cours)
'Si médecin peut faire l'astreinte, on affiche les initiales et mise en couleur en fonction préférences de jour de semaine
u = 1
For i = 2 To 20
If G_sh(3).Cells(i, 3) <> "" Then
   initiales = G_sh(3).Range("C" & i).Value
   Set C = G_sh(1).Range("AV" & row_select & ":BC" & row_select).Find(What:=initiales, LookIn:=xlValues)
   Set A = G_sh(1).Range("V" & Row_Previous_Saturday & ":W" & Row_Next_Sunday).Find(What:=initiales, LookIn:=xlValues)
    Set B = Nothing
    If UserFormAstreintes.CheckBox1.Value = True Then
       Set B = G_sh(1).Range("BL" & row_select & ":BS" & row_select).Find(What:=initiales, LookIn:=xlValues)
    End If
   If C Is Nothing And A Is Nothing And B Is Nothing Then
        UserFormAstreintes.Controls("commandbutton" & u).Caption = initiales
        Set range_initiales = G_sh(3).Columns(3).Cells.Find(What:=initiales)
        pref = G_sh(3).Cells(range_initiales.Row, joursemaine + 5).Value

            If pref Like "Oui" Then
                UserFormAstreintes.Controls("commandbutton" & u).BackColor = RGB(0, 255, 0)
            ElseIf pref = "Non" Then
                UserFormAstreintes.Controls("commandbutton" & u).BackColor = RGB(255, 0, 0)
            ElseIf pref = "OK" Then
                UserFormAstreintes.Controls("commandbutton" & u).BackColor = RGB(255, 255, 0)
            End If

        UserFormAstreintes.Controls("textboxast" & u).Value = G_sh(8).Range("C" & range_initiales.Row).Value
        u = u + 1
    End If
End If
Next i
End If

fin:
Start_protect
Application.ScreenUpdating = True
End Sub
